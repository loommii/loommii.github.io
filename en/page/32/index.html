package main

import "fmt"

func app() func(string) string {
    t := "Hi"
    c := func(b string) string {
        t = t + " " + b
        return t
    }
    return c
}

func main() {
    a := app()
    b := app()
    a("go")
    fmt.Println(b("All"))
    fmt.Println(a("All"))
}

A: Hi All; B: Hi go All; C: Hi; D: go Al

🔑 Answer Analysis:

Closure fmt.Println(b("All")) outputs A (Hi All) fmt.Println(a("All")) outputs B (Hi go All)
